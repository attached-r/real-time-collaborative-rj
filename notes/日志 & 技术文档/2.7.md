## 今日关键代码片段

### 登录页（部分）

```ts
const handleLogin = async () => {
  try {
    const res = await api.post("/api/auth/login", { username, password });
    localStorage.setItem("token", res.data.token);
    toast.success("登录成功！");
    router.push("/documents");
  } catch (err) {
    toast.error("登录失败，请检查用户名或密码");
  }
};
```

### 注册页（核心逻辑）

```ts
const handleRegister = async () => {
  if (form.password !== form.confirmPassword) {
    toast.error("两次密码不一致");
    return;
  }
  try {
    await api.post("/api/auth/register", form);
    toast.success("注册成功！请登录");
    router.push("/login");
  } catch (err) {
    toast.error(err.response?.data?.message || "注册失败");
  }
};
```

### 文档列表（loading + toast + 新建）

```ts
const createDoc = async () => {
  loading.value = true;
  try {
    const res = await api.post("/api/documents", { title, content });
    toast.success(`创建成功！ID: ${res.data.id}`);
    await fetchDocuments();
  } catch (err) {
    toast.error("创建失败：" + (err.response?.data?.message || "未知错误"));
  } finally {
    loading.value = false;
  }
};
```
